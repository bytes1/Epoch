export async function createApp(options: { db?: ReturnType<typeof createDatabase>; genesis?: GenesisService; resolver?: ResolutionExecutor } = {}) {
  const owned = options.db ?? createDatabase();
  const app = Fastify({ logger: true });
  const ticker = new TickerHub();
  await app.register(cors, { origin: process.env.CORS_ORIGIN ?? true });
  const chipnet = process.env.BCH_NETWORK === 'chipnet';
  const genesisService = options.genesis ?? new GenesisService(owned.db, chipnet
    ? (process.env.MARKET_WALLET_PRIVATE_KEY ? new ChipnetGenesisAdapter() : { async createMarket() { throw new Error('Set MARKET_WALLET_PRIVATE_KEY to enable Chipnet market genesis.'); } })
    : new MockGenesisAdapter());
  const chipnetReady = Boolean(process.env.MARKET_WALLET_PRIVATE_KEY && (process.env.ORACLE_PRIVATE_KEY || process.env.MARKET_WALLET_PRIVATE_KEY));
  const matcherExecutor = chipnet ? (chipnetReady ? new ChipnetMatchExecutor(owned.db) : new UnconfiguredMatchExecutor()) : new MockMatchExecutor();
  const matcher = new MatcherService(owned.db, matcherExecutor, (marketId, update) => ticker.publish(marketId, update));

  await registerOrderRoutes(app, owned.db, matcher, chipnet ? new ChipnetOrderVerifier() : undefined);
  app.addHook('onClose', async () => { if (!options.db) owned.close(); });
  return app;
}

export function createDatabase(path = process.env.DATABASE_PATH ?? './data/market.db') {
  if (path !== ':memory:') {
    mkdirSync(dirname(resolve(path)), { recursive: true });
  }
  const sqlite = new Database(path);
  if (path !== ':memory:') sqlite.pragma('journal_mode = WAL');
  sqlite.pragma('foreign_keys = ON');
  createTables(sqlite);
  return { sqlite, db: drizzle(sqlite, { schema }), close: () => sqlite.close() };
}

import { and, asc, eq, inArray } from 'drizzle-orm';
import type { FastifyInstance } from 'fastify';
import { z } from 'zod';
import type { MarketDb } from '../db/index.js';
import { markets, orders, trades } from '../db/schema.js';
import { GenesisService } from '../services/genesis.service.js';

const createMarket = z.object({ title: z.string().min(3).max(200), description: z.string().max(5000).optional() });

export async function registerMarketRoutes(app: FastifyInstance, db: MarketDb, genesis: GenesisService,
  resolver: ResolutionExecutor = new UnconfiguredResolutionExecutor()) {
  app.post('/api/markets', async (request, reply) => {
    if (process.env.ADMIN_API_KEY && request.headers['x-admin-key'] !== process.env.ADMIN_API_KEY) return reply.code(401).send({ error: 'Unauthorized' });
    const parsed = createMarket.safeParse(request.body);
    if (!parsed.success) return reply.code(400).send({ error: parsed.error.flatten() });
    return reply.code(201).send(await genesis.create(parsed.data.title, parsed.data.description));
  });

  app.get('/api/markets', async () => db.select().from(markets).all().map((m) => ({
    ...m, yesProbabilityPercent: m.lastYesPriceSats / 10, noProbabilityPercent: m.lastNoPriceSats / 10,
  })));

  app.get('/api/markets/:id/orderbook', async (request, reply) => {
    const { id } = request.params as { id: string };
    const market = db.select().from(markets).where(eq(markets.id, id)).get();
    if (!market) return reply.code(404).send({ error: 'Market not found' });
    const rows = db.select().from(orders).where(and(eq(orders.marketId, id), inArray(orders.status, ['OPEN', 'PARTIAL']))).all();
    const group = (side: 'BUY' | 'SELL', outcome: 'YES' | 'NO') => {
      const grouped = new Map<number, number>();
      for (const order of rows.filter((x) => x.side === side && x.outcome === outcome))
        grouped.set(order.priceSats, (grouped.get(order.priceSats) ?? 0) + order.remainingShares);
      return [...grouped].map(([priceSats, shares]) => ({ priceSats, shares }))
        .sort((a, b) => side === 'BUY' ? b.priceSats - a.priceSats : a.priceSats - b.priceSats);
    };
    return { yes: { bids: group('BUY', 'YES'), asks: group('SELL', 'YES') }, no: { bids: group('BUY', 'NO'), asks: group('SELL', 'NO') },
      ticks: db.select().from(trades).where(eq(trades.marketId, id)).orderBy(asc(trades.executedAt)).all() };
  });

  app.post('/api/markets/:id/resolve', async (request, reply) => {
    if (process.env.ADMIN_API_KEY && request.headers['x-admin-key'] !== process.env.ADMIN_API_KEY) return reply.code(401).send({ error: 'Unauthorized' });
    const { id } = request.params as { id: string };
    const parsed = z.object({ outcome: z.enum(['YES', 'NO']) }).safeParse(request.body);
    if (!parsed.success) return reply.code(400).send({ error: parsed.error.flatten() });
    const market = db.select().from(markets).where(eq(markets.id, id)).get();
    if (!market) return reply.code(404).send({ error: 'Market not found' });
    if (market.status !== 'OPEN') return reply.code(409).send({ error: 'Market already resolved' });
    const oracleWif = process.env.ORACLE_PRIVATE_KEY || process.env.MARKET_WALLET_PRIVATE_KEY;
    if (!oracleWif) return reply.code(503).send({ error: 'Set ORACLE_PRIVATE_KEY or MARKET_WALLET_PRIVATE_KEY in .env to authorize resolution.' });
    const run = async () => {
      const status = parsed.data.outcome === 'YES' ? 'RESOLVED_YES' : 'RESOLVED_NO';
      let signature: string;
      try { signature = signResolution(market.marketIdHex, parsed.data.outcome, oracleWif); }
      catch (error) { return reply.code(400).send({ error: error instanceof Error ? error.message : 'Oracle signing failed' }); }
      let txid: string;
      try { txid = await resolver.resolve(market, parsed.data.outcome, signature); }
      catch (error) { return reply.code(503).send({ error: error instanceof Error ? error.message : 'Resolution transaction failed' }); }
      db.update(markets).set({ status, resolvedAt: Date.now() }).where(eq(markets.id, id)).run();
      return { marketId: market.marketIdHex, outcome: parsed.data.outcome, status, signature, txid };
    };
    return marketMutex.runExclusive(id, run);
  });
}

export interface OrderVerifier {

    utxoVout: number; utxoSatoshis: number; tokenCategory?: string; tokenAmount?: number; priceSats: number; shares: number }): Promise<void>;
}

const MAX_SAFE_SHARES = Math.floor(Number.MAX_SAFE_INTEGER / 1000);
const prepareSchema = z.object({ marketId: z.string(), side: z.enum(['BUY', 'SELL']), outcome: z.enum(['YES', 'NO']),
  priceSats: z.number().int().min(1).max(999), shares: z.number().int().positive().max(MAX_SAFE_SHARES), makerPubKey: z.string().regex(/^(02|03)[0-9a-f]{64}$/i) });
const registerSchema = prepareSchema.extend({ makerAddress: z.string().min(2), utxoTxid: z.string().regex(/^[0-9a-f]{64}$/i),
  utxoVout: z.number().int().nonnegative().max(Number.MAX_SAFE_INTEGER), utxoSatoshis: z.number().int().positive().max(Number.MAX_SAFE_INTEGER), tokenCategory: z.string().optional(), tokenAmount: z.number().int().nonnegative().max(MAX_SAFE_SHARES).optional() });

export async function registerOrderRoutes(app: FastifyInstance, db: MarketDb, matcher: MatcherService, verifier?: OrderVerifier) {
  app.post('/api/orders/prepare', async (request, reply) => {
    const parsed = prepareSchema.safeParse(request.body);
    if (!parsed.success) return reply.code(400).send({ error: parsed.error.flatten() });
    const input = parsed.data;
    const market = db.select().from(markets).where(eq(markets.id, input.marketId)).get();
    if (!market) return reply.code(404).send({ error: 'Market not found' });
    if (market.status !== 'OPEN') return reply.code(409).send({ error: 'Market is closed' });
    const requiredSatoshis = input.side === 'BUY' ? input.priceSats * input.shares + 1000 : 1000;
    const pubkeyHash = createHash('ripemd160').update(createHash('sha256').update(Buffer.from(input.makerPubKey, 'hex')).digest()).digest('hex');
    const p2pkhBytecode = '76a914' + pubkeyHash + '88ac';
    const makerLockingBytecodeHash = createHash('ripemd160').update(createHash('sha256').update(Buffer.from(p2pkhBytecode, 'hex')).digest()).digest('hex');
    const contractName = input.side === 'BUY' ? 'BuyOrder' : 'SellOrder';
    const artifact = JSON.parse(readFileSync(new URL('../../../contracts/artifacts/' + contractName + '.json', import.meta.url), 'utf8'));
    const contract = new Contract(artifact, [input.makerPubKey, makerLockingBytecodeHash,
      input.outcome === 'YES' ? market.yesCategoryId : market.noCategoryId, BigInt(input.priceSats)], { provider: new MockNetworkProvider() });
    return { contract: contractName, address: input.side === 'SELL' ? contract.tokenAddress : contract.address,
      requiredSatoshis, requiredTokenCategory: input.side === 'SELL' ? (input.outcome === 'YES' ? market.yesCategoryId : market.noCategoryId) : null,
      requiredTokenAmount: input.side === 'SELL' ? input.shares : 0,
      makerLockingBytecodeHash, constructorArgs: [input.makerPubKey, makerLockingBytecodeHash,
        input.outcome === 'YES' ? market.yesCategoryId : market.noCategoryId, input.priceSats] };
  });

  app.post('/api/orders/register', async (request, reply) => {
    const parsed = registerSchema.safeParse(request.body);
    if (!parsed.success) return reply.code(400).send({ error: parsed.error.flatten() });
    const input = parsed.data;
    const market = db.select().from(markets).where(eq(markets.id, input.marketId)).get();
    if (!market) return reply.code(404).send({ error: 'Market not found' });
    if (market.status !== 'OPEN') return reply.code(409).send({ error: 'Market is closed' });
    const alreadyRegistered = db.select().from(orders).where(and(eq(orders.utxoTxid, input.utxoTxid), eq(orders.utxoVout, input.utxoVout))).get();
    if (alreadyRegistered) return reply.code(409).send({ error: 'This UTXO is already registered as an order.' });
    const expectedCategory = input.outcome === 'YES' ? market.yesCategoryId : market.noCategoryId;
    const expectedSats = input.side === 'BUY' ? input.shares * input.priceSats + 1000 : 1000;
    if (input.side === 'BUY' && input.utxoSatoshis !== expectedSats) return reply.code(400).send({ error: 'Buy order UTXO must contain exactly bid × shares plus 1,000 sats.' });
    if (input.side === 'SELL' && input.utxoSatoshis !== 1000) return reply.code(400).send({ error: 'Sell order UTXO must contain exactly 1,000 sats.' });
    if (input.side === 'SELL' && (input.tokenCategory !== expectedCategory || input.tokenAmount !== input.shares))
      return reply.code(400).send({ error: 'Order UTXO does not contain the required outcome tokens' });
    if (verifier) {
      try { await verifier.verify({ market, side: input.side, outcome: input.outcome, makerPubKey: input.makerPubKey, utxoTxid: input.utxoTxid,
        utxoVout: input.utxoVout, utxoSatoshis: input.utxoSatoshis, tokenCategory: input.tokenCategory, tokenAmount: input.tokenAmount,
        priceSats: input.priceSats, shares: input.shares }); }
      catch (error) { return reply.code(400).send({ error: error instanceof Error ? error.message : 'Order UTXO verification failed' }); }
    }
    const id = randomUUID();
    const matches = await matcher.register({ id, marketId: input.marketId, makerAddress: input.makerAddress, makerPubkey: input.makerPubKey,
      side: input.side, outcome: input.outcome, priceSats: input.priceSats, totalShares: input.shares, remainingShares: input.shares,
      utxoTxid: input.utxoTxid, utxoVout: input.utxoVout, utxoSatoshis: input.utxoSatoshis, status: 'OPEN', createdAt: Date.now() });
    return reply.code(201).send({ orderId: id, matches });
  });
}

const artifact = (name: string) => JSON.parse(readFileSync(new URL('../../../contracts/artifacts/' + name + '.json', import.meta.url), 'utf8'));
const hash160 = (bytes: Uint8Array) => createHash('ripemd160').update(createHash('sha256').update(bytes).digest()).digest('hex');
const toHex = (value: string | Uint8Array) => typeof value === 'string' ? value : Buffer.from(value).toString('hex');

function addOutputAt(tx: TransactionBuilder, index: number, output: Parameters<TransactionBuilder['addOutput']>[0], paddingAddress: string) {
  while (tx.outputs.length < index) tx.addOutput({ to: paddingAddress, amount: 1000n });
  if (tx.outputs.length !== index) throw new Error(`Transaction output layout expected index ${index}, found ${tx.outputs.length}.`);
  tx.addOutput(output);
}

export function makerTokenAddress(pubkeyHex: string) {
  const result = publicKeyToP2pkhCashAddress({ publicKey: Buffer.from(pubkeyHex, 'hex'), prefix: 'bchtest', tokenSupport: true });
  if (typeof result === 'string') throw new Error('Could not derive maker address: ' + result);
  return result.address;
}

function orderContract(order: BookOrder, market: typeof markets.$inferSelect, provider: ElectrumNetworkProvider) {
  const contractName = order.side === 'BUY' ? 'BuyOrder' : 'SellOrder';
  const pubkey = Buffer.from(order.makerPubkey, 'hex');
  const p2pkh = Buffer.from('76a914' + hash160(pubkey) + '88ac', 'hex');
  const args = [order.makerPubkey, hash160(p2pkh), order.outcome === 'YES' ? market.yesCategoryId : market.noCategoryId, BigInt(order.priceSats)];
  return new Contract(artifact(contractName), args, { provider });
}

function marketVaultArtifact(market: typeof markets.$inferSelect, oraclePubkey: Uint8Array, provider: ElectrumNetworkProvider) {
  for (const name of ['MarketVault', 'MarketVaultV1']) {
    const contract = new Contract(artifact(name), [market.yesCategoryId, market.noCategoryId, oraclePubkey, market.marketIdHex], { provider });
    if (contract.tokenAddress === market.vaultContractAddress) return { name, contract };
  }
  throw new Error('No known MarketVault artifact matches this market address. Preserve its deployment artifact before upgrading contracts.');
}

  private readonly provider = new ElectrumNetworkProvider('chipnet');
  async verify(input: { market: typeof markets.$inferSelect; side: 'BUY' | 'SELL'; outcome: 'YES' | 'NO'; makerPubKey: string; utxoTxid: string; utxoVout: number; utxoSatoshis: number; tokenCategory?: string; tokenAmount?: number; priceSats: number; shares: number }) {
    const token = input.outcome === 'YES' ? input.market.yesCategoryId : input.market.noCategoryId;
    const order = { makerPubkey: input.makerPubKey, side: input.side, outcome: input.outcome, priceSats: input.priceSats } as BookOrder;
    const contract = orderContract(order, input.market, this.provider);
    const address = input.side === 'SELL' ? contract.tokenAddress : contract.address;
    const utxo = (await this.provider.getUtxos(address)).find((x) => x.txid.toLowerCase() === input.utxoTxid.toLowerCase() && x.vout === input.utxoVout);
    if (!utxo) throw new Error('The submitted order outpoint is not an unspent output of its order contract.');
    if (Number(utxo.satoshis) !== input.utxoSatoshis) throw new Error('Submitted BCH amount does not match the on-chain order UTXO.');
    if (input.side === 'BUY') {
      if (utxo.token || utxo.satoshis !== BigInt(input.priceSats * input.shares + 1000)) throw new Error('Buy order UTXO must contain exactly bid × shares plus 1,000 sats.');
    } else if (utxo.satoshis !== 1000n || !utxo.token || utxo.token.category.toLowerCase() !== token.toLowerCase() || utxo.token.amount !== BigInt(input.shares)) {
      throw new Error('Sell order UTXO must contain the exact outcome category and share amount.');
    }
  }
}

  private async dispatch(marketId: string, kind: MatchType, shares: number, matched: [BookOrder, BookOrder]) {

    if (order.side === 'BUY' && (utxo.token || utxo.satoshis !== BigInt(order.remainingShares * order.priceSats + 1000)))
      throw new Error(`Buy order ${order.id} does not have the exact remaining BCH collateral.`);

      .sort((x, y) => Number(y.satoshis - x.satoshis));
    if (!candidates[0]) throw new Error('Chipnet match requires a native BCH wallet UTXO of at least 5,000 sats to pay the transaction fee.');
    return candidates[0];

    const utxos: Utxo[] = [];

      const address = row.utxoRole === 'VAULT_MASTER' ? vaultContract.tokenAddress : baton.tokenAddress;

      const found = addressUtxos.find((u) => u.txid.toLowerCase() === row.txid.toLowerCase() && u.vout === row.vout) ??

        .where(eq(vaultUtxos.id, row.id)).run();

  private async mint(market: typeof markets.$inferSelect, shares: number, a: BookOrder, b: BookOrder) {

    const [{ rows, utxos }, yesLive, noLive, fee] = await Promise.all([
      this.vaultState(market.id, ['VAULT_MASTER', 'YES_BATON', 'NO_BATON']), this.liveOrder(yes, market), this.liveOrder(no, market), this.feeInput(),

    tx.addInput(yesLive.utxo, (yesLive.contract.unlock as any).fill(BigInt(shares)));

      { to: this.payout(yes), amount: 1000n, token: { category: market.yesCategoryId, amount: BigInt(shares) } },
      { to: this.payout(no), amount: 1000n, token: { category: market.noCategoryId, amount: BigInt(shares) } },
    ]);
    const yesLeft = yes.remainingShares > shares;
    const noLeft = no.remainingShares > shares;
    if (yesLeft) addOutputAt(tx, continuationVout('MINT', yes), { to: yesLive.contract.address, amount: yesLive.utxo.satoshis - BigInt(shares * yes.priceSats) }, this.walletAddress);
    if (noLeft) addOutputAt(tx, continuationVout('MINT', no), { to: noLive.contract.address, amount: noLive.utxo.satoshis - BigInt(shares * no.priceSats) }, this.walletAddress);
    const yesExecution = yesExecutionPrice('MINT', yes, no);
    const yesRefund = shares * (yes.priceSats - yesExecution);
    const noRefund = shares * (no.priceSats - (1000 - yesExecution));
    if (yesRefund > 0) tx.addOutput({ to: this.payout(yes), amount: BigInt(yesRefund) });
    if (noRefund > 0) tx.addOutput({ to: this.payout(no), amount: BigInt(noRefund) });

  }

  private async merge(market: typeof markets.$inferSelect, shares: number, a: BookOrder, b: BookOrder) {
    const yes = a.outcome === 'YES' ? a : b; const no = a.outcome === 'NO' ? a : b;
    const [{ rows, utxos }, yesLive, noLive, fee] = await Promise.all([
      this.vaultState(market.id, ['VAULT_MASTER']), this.liveOrder(yes, market), this.liveOrder(no, market), this.feeInput(),
    ]);
    const vault = this.vaultContract(market);
    const yesExecution = yesExecutionPrice('MERGE', yes, no);
    const noExecution = 1000 - yesExecution;
    const tx = new TransactionBuilder({ provider: this.provider, maximumFeeSatoshis: 30_000n, allowImplicitFungibleTokenBurn: true });
    tx.addInput(utxos[0]!, (vault.unlock as any).mergeCompleteSet(BigInt(shares)));
    tx.addInput(yesLive.utxo, (yesLive.contract.unlock as any).fill(BigInt(shares), BigInt(yesExecution)));
    tx.addInput(noLive.utxo, (noLive.contract.unlock as any).fill(BigInt(shares), BigInt(noExecution)));
    tx.addInput(fee, this.signer.unlockP2PKH());
    tx.addOutput({ to: vault.tokenAddress, amount: utxos[0]!.satoshis - BigInt(shares * 1000), token: { category: market.vaultCategoryId, amount: 0n, nft: { capability: 'mutable', commitment: '00' } } });
    tx.addOutput({ to: this.payout(yes), amount: BigInt(shares * yesExecution) });
    tx.addOutput({ to: this.payout(no), amount: BigInt(shares * noExecution) });
    const yesLeft = yes.remainingShares > shares;
    const noLeft = no.remainingShares > shares;
    if (yesLeft) addOutputAt(tx, continuationVout('MERGE', yes), { to: yesLive.contract.tokenAddress, amount: 1000n, token: { category: market.yesCategoryId, amount: BigInt(yes.remainingShares - shares) } }, this.walletAddress);
    if (noLeft) addOutputAt(tx, continuationVout('MERGE', no), { to: noLive.contract.tokenAddress, amount: 1000n, token: { category: market.noCategoryId, amount: BigInt(no.remainingShares - shares) } }, this.walletAddress);
    if (!yesLeft) tx.addOutput({ to: this.payout(yes), amount: 1000n });
    if (!noLeft) tx.addOutput({ to: this.payout(no), amount: 1000n });
    tx.addBchChangeOutputIfNeeded({ to: this.walletAddress, feeRate: 1 });
    const sent = await tx.send();
    this.updateVault(rows, sent.txid, [{ vout: 0, satoshis: Number(sent.outputs[0]!.valueSatoshis) }]);
    return sent.txid;
  }

  private updateVault(rows: Array<typeof vaultUtxos.$inferSelect>, txid: string, outputs: Array<{ vout: number; satoshis: number }>) {
    const now = Date.now();
    this.db.transaction(() => rows.forEach((row, index) => this.db.update(vaultUtxos).set({ txid, vout: outputs[index]!.vout,
      satoshis: outputs[index]!.satoshis, updatedAt: now }).where(eq(vaultUtxos.id, row.id)).run()));
  }

/** Builds and broadcasts the three category-seeding transactions and the fixed 0/1/2 vault bundle. */
export class ChipnetGenesisAdapter implements GenesisAdapter {
  private readonly provider = new ElectrumNetworkProvider('chipnet');
  private readonly privateKey = decodeChipnetWallet();
  private readonly signer = new SignatureTemplate(this.privateKey);
  private readonly walletAddress = chipnetFundingAddress(this.privateKey);

  }

  private async createMarketExclusive(): Promise<GenesisResult> {
    const walletUtxos = (await this.provider.getUtxos(this.walletAddress))
      .filter((utxo) => !utxo.token).sort((a, b) => Number(b.satoshis - a.satoshis));
    const initialFunding: Utxo[] = [];
    let fundingSatoshis = 0n;
    for (const utxo of walletUtxos) {
      initialFunding.push(utxo);
      fundingSatoshis += utxo.satoshis;
      if (fundingSatoshis >= 6_000n) break;
    }
    if (fundingSatoshis < 6_000n) throw new Error('Fund the Chipnet wallet with at least 6,000 sats of native BCH, then retry market creation.');

    const oracleWif = process.env.ORACLE_PRIVATE_KEY || process.env.MARKET_WALLET_PRIVATE_KEY;
    if (!oracleWif) throw new Error('Set ORACLE_PRIVATE_KEY or MARKET_WALLET_PRIVATE_KEY before creating a market.');
    const oracleKey = decodePrivateKeyWif(oracleWif);
    if (typeof oracleKey === 'string') throw new Error('Invalid oracle WIF: ' + oracleKey);
    const oraclePubKey = secp256k1.derivePublicKeyCompressed(oracleKey.privateKey);
    if (typeof oraclePubKey === 'string') throw new Error('Could not derive the oracle public key: ' + oraclePubKey);
    const marketIdHex = randomBytes(4).toString('hex');

    const preGenesis = [] as Array<{ txid: string; vout: number }>;
    let spendable = initialFunding;
    for (let index = 0; index < 3; index += 1) {
      const tx = new TransactionBuilder({ provider: this.provider, maximumFeeSatoshis: 10_000n });
      for (const utxo of spendable) tx.addInput(utxo, this.signer.unlockP2PKH());
      tx.addOutput({ to: this.walletAddress, amount: 1000n });
      tx.addBchChangeOutputIfNeeded({ to: this.walletAddress, feeRate: 1 });
      const sent = await tx.send();
      preGenesis.push({ txid: sent.txid, vout: 0 });
      const change = sent.outputs[1]?.valueSatoshis;
      if (change === undefined) throw new Error('The wallet UTXO was too small to preserve BCH change for the next genesis transaction.');
      const changeSatoshis = typeof change === 'bigint' ? change : typeof change === 'number' ? BigInt(change) : binToValueSatoshis(change);
      if (changeSatoshis < 1000n) throw new Error('Insufficient BCH remains after category seeding; fund the wallet with more than 6,000 sats and retry.');
      spendable = [{ txid: sent.txid, vout: 1, satoshis: changeSatoshis }];
    }

    const yesCategoryId = preGenesis[0]!.txid;
    const noCategoryId = preGenesis[1]!.txid;
    const vaultCategoryId = preGenesis[2]!.txid;

    const vault = new Contract(readArtifact('MarketVault'), [yesCategoryId, noCategoryId, oraclePubKey, marketIdHex], { provider: this.provider });
    const vaultScriptHash = hash160(vault.lockingBytecode);
    const yesBaton = new Contract(readArtifact('MintingBaton'), [vaultScriptHash], { provider: this.provider });
    const noBaton = new Contract(readArtifact('MintingBaton'), [vaultScriptHash], { provider: this.provider });
    const genesis = new TransactionBuilder({ provider: this.provider, maximumFeeSatoshis: 10_000n });
    for (const outpoint of preGenesis) genesis.addInput({ ...outpoint, satoshis: 1000n }, this.signer.unlockP2PKH());
    genesis.addInput(spendable[0]!, this.signer.unlockP2PKH());
    genesis.addOutputs([
      { to: vault.tokenAddress, amount: 1000n, token: { category: vaultCategoryId, amount: 0n, nft: { capability: 'mutable', commitment: '00' } } },
      { to: yesBaton.tokenAddress, amount: 1000n, token: { category: yesCategoryId, amount: 0n, nft: { capability: 'minting', commitment: '' } } },
      { to: noBaton.tokenAddress, amount: 1000n, token: { category: noCategoryId, amount: 0n, nft: { capability: 'minting', commitment: '' } } },
    ]);
    genesis.addBchChangeOutputIfNeeded({ to: this.walletAddress, feeRate: 1 });
    const sent = await genesis.send();

    return { marketIdHex, yesCategoryId, noCategoryId, vaultCategoryId,
      vaultContractAddress: vault.tokenAddress, txid: sent.txid };
  }

export type RegisteredOrder = typeof orders.$inferInsert;
export type MatchType = 'DIRECT' | 'MINT' | 'MERGE';
export type BookOrder = typeof orders.$inferSelect;
export interface MatchExecutor { execute(marketId: string, kind: MatchType, shares: number, matchedOrders: [BookOrder, BookOrder]): Promise<string> }
export class MockMatchExecutor implements MatchExecutor {

/** YES-side execution price under price-time priority. */
export function yesExecutionPrice(kind: MatchType, a: BookOrder, b: BookOrder) {
  if (kind === 'MINT') {
    const yes = a.outcome === 'YES' ? a : b;
    const no = a.outcome === 'NO' ? a : b;
    const maker = a.createdAt <= b.createdAt ? a : b;
    return maker.outcome === 'YES' ? yes.priceSats : 1000 - no.priceSats;
  }
  if (kind === 'MERGE') {
    const maker = a.createdAt <= b.createdAt ? a : b;
    return maker.outcome === 'YES' ? maker.priceSats : 1000 - maker.priceSats;
  }
  const maker = a.createdAt <= b.createdAt ? a : b;
  return maker.outcome === 'YES' ? maker.priceSats : 1000 - maker.priceSats;
}

export function continuationVout(type: MatchType, order: BookOrder) {
  if (type === 'DIRECT') return order.side === 'BUY' ? 2 : 3;
  if (type === 'MINT') return order.outcome === 'YES' ? 5 : 6;
  return order.outcome === 'YES' ? 3 : 4;
}

/** Price-time matching with serialized vault-backed mint/merge execution. */
export class MatcherService {
  constructor(private readonly db: MarketDb, private readonly executor: MatchExecutor = new MockMatchExecutor(),
    private readonly publish: (marketId: string, update: unknown) => void = () => {}) {}

  async register(order: RegisteredOrder) {
    this.db.insert(orders).values(order).run();
    return this.match(order.marketId);
  }

  async match(marketId: string) {
    return marketMutex.runExclusive(marketId, async () => {
      const result = [] as Array<{ type: MatchType; shares: number; txid: string }>;
      while (true) {
        const open = this.db.select().from(orders).where(and(eq(orders.marketId, marketId), inArray(orders.status, ['OPEN', 'PARTIAL']))).all();
        const next = this.findMatch(open);
        if (!next) break;
        const txid = await this.executor.execute(marketId, next.type, next.shares, [next.a, next.b]);
        const yesPrice = yesExecutionPrice(next.type, next.a, next.b);
        const noPrice = 1000 - yesPrice;
        this.db.transaction(() => {
          for (const order of [next.a, next.b]) {
            const left = order.remainingShares - next.shares;
            const continuationOutput = left === 0 ? undefined : continuationVout(next.type, order);
            const remainingSatoshis = order.side === 'BUY'
              ? order.utxoSatoshis - next.shares * order.priceSats
              : Math.max(1000, order.utxoSatoshis);
            this.db.update(orders).set({ remainingShares: left, status: left === 0 ? 'FILLED' : 'PARTIAL',
              ...(continuationOutput === undefined ? {} : { utxoTxid: txid, utxoVout: continuationOutput, utxoSatoshis: remainingSatoshis }) })
              .where(eq(orders.id, order.id)).run();
          }
          this.db.insert(trades).values({ id: randomUUID(), marketId, matchType: next.type, yesPriceSats: yesPrice,
            noPriceSats: noPrice, sharesTraded: next.shares, bchTxid: txid, executedAt: Date.now() }).run();
          this.db.update(markets).set({ lastYesPriceSats: yesPrice, lastNoPriceSats: noPrice,
            volumeSats: sql`${markets.volumeSats} + ${next.shares * 1000}` }).where(eq(markets.id, marketId)).run();
        });
        const update = { type: next.type, shares: next.shares, yesPriceSats: yesPrice, noPriceSats: noPrice, txid };
        this.publish(marketId, update);
        result.push({ type: next.type, shares: next.shares, txid });
      }
      return result;
    });
  }

  private findMatch(rows: Array<typeof orders.$inferSelect>) {
    const candidates = rows.filter((row) => row.remainingShares > 0);
    const buys = candidates.filter((x) => x.side === 'BUY').sort((a, b) => b.priceSats - a.priceSats || a.createdAt - b.createdAt);
    const sells = candidates.filter((x) => x.side === 'SELL').sort((a, b) => a.priceSats - b.priceSats || a.createdAt - b.createdAt);
    for (const bid of buys) {
      for (const ask of sells) if (bid.outcome === ask.outcome && bid.priceSats >= ask.priceSats)
        if (Math.min(bid.remainingShares, ask.remainingShares) * ask.priceSats >= 1000)
          return { type: 'DIRECT' as const, a: bid, b: ask, shares: Math.min(bid.remainingShares, ask.remainingShares) };
    }
    for (const yes of buys.filter((x) => x.outcome === 'YES')) for (const no of buys.filter((x) => x.outcome === 'NO')) {
      if (yes.priceSats + no.priceSats >= 1000) return { type: 'MINT' as const, a: yes, b: no, shares: Math.min(yes.remainingShares, no.remainingShares) };
    }
    const yesAsks = sells.filter((x) => x.outcome === 'YES'); const noAsks = sells.filter((x) => x.outcome === 'NO');
    for (const yes of yesAsks) for (const no of noAsks) {
      const shares = Math.min(yes.remainingShares, no.remainingShares);
      if (yes.priceSats + no.priceSats <= 1000 && shares * yes.priceSats >= 1000 && shares * no.priceSats >= 1000)
        return { type: 'MERGE' as const, a: yes, b: no, shares };
    }
    return undefined;
  }

}

/** BCH OP_CHECKDATASIG signs SHA256(market_id || outcome_byte) with a DER ECDSA signature. */
export function signResolution(marketIdHex: string, outcome: 'YES' | 'NO', privateKeyWif: string) {
  const decoded = decodePrivateKeyWif(privateKeyWif);
  if (typeof decoded === 'string') throw new Error('Invalid ORACLE_PRIVATE_KEY: ' + decoded);
  if (!/^[0-9a-f]{8}$/i.test(marketIdHex)) throw new Error('marketIdHex must be exactly 4 bytes');
  const message = Buffer.from(marketIdHex + (outcome === 'YES' ? '01' : '02'), 'hex');
  const signature = secp256k1.signMessageHashDER(decoded.privateKey, sha256.hash(message));
  if (typeof signature === 'string') throw new Error('Oracle signing failed: ' + signature);
  return Buffer.from(signature).toString('hex');
}

/** Serializes zero-confirmation spends per market without an external queue. */
export class KeyedMutex {
  private readonly tails = new Map<string, Promise<void>>();

  async runExclusive<T>(key: string, operation: () => Promise<T>): Promise<T> {
    const previous = this.tails.get(key) ?? Promise.resolve();
    let release!: () => void;
    const current = new Promise<void>((resolve) => { release = resolve; });
    this.tails.set(key, current);
    await previous;
    try { return await operation(); }
    finally {
      release();
      if (this.tails.get(key) === current) this.tails.delete(key);
    }
  }
}

export const marketMutex = new KeyedMutex();
/** Serializes spends from the shared operational wallet across all markets. */
export const walletMutex = new KeyedMutex();

pragma cashscript ^0.13.0;

contract SellOrder(pubkey makerPubKey, bytes20 makerLockingBytecodeHash, bytes32 tokenCategory, int pricePerShareSats) {
    function fill(int sharesFilled, int executionPriceSats) {
        require(sharesFilled > 0);
        require(pricePerShareSats > 0 && pricePerShareSats < 1000);
        require(executionPriceSats >= pricePerShareSats && executionPriceSats < 1000);
        require(tx.inputs[this.activeInputIndex].tokenCategory == tokenCategory);
        require(sharesFilled <= tx.inputs[this.activeInputIndex].tokenAmount);
        int payoutSats = sharesFilled * executionPriceSats;
        require(tx.outputs[this.activeInputIndex].value == payoutSats);
        require(hash160(tx.outputs[this.activeInputIndex].lockingBytecode) == makerLockingBytecodeHash);
        if (sharesFilled < tx.inputs[this.activeInputIndex].tokenAmount) {
            require(tx.outputs[this.activeInputIndex + 2].tokenCategory == tokenCategory);
            require(tx.outputs[this.activeInputIndex + 2].tokenAmount == tx.inputs[this.activeInputIndex].tokenAmount - sharesFilled);
            require(tx.outputs[this.activeInputIndex + 2].lockingBytecode == tx.inputs[this.activeInputIndex].lockingBytecode);
        }
    }

    function cancel(sig makerSig) {
        require(checkSig(makerSig, makerPubKey));
    }
}
